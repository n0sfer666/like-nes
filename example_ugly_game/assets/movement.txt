# читается `ProfileTable` БЕЗ парсинга, прямо из mmap-региона.
#
# ПОТРЕБИТЕЛЯ В РАНТАЙМЕ ПОКА НЕТ: контроллер по-прежнему берёт значения у `default_profile()`, и
# правка этого файла сама по себе игру не меняет. Доехали пекарь и читатель; подключение приходит с
# образцом-платформером. Пока источник правды — код, и сверка ниже направлена именно так.
#
# Единицы — пиксель и секунда, +Y ВНИЗ (physics/units.hpp). Окна прощения — в ТИКАХ при 60 Гц:
# окно в секундах при делении на шаг даёт дробное число тиков, а округление дроби есть поведение,
# зависящее от шага, то есть от машины.
#
# Значения ОБЯЗАНЫ совпадать с `default_profile()` в `profile.cpp`, и это утверждение гейта:
# `framework_character_bundle_test` читает секцию из готового game.bundle и сверяет её с ним поле в
# поле. Правка одних чисел здесь красит CI — это не помеха будущему подбору, а следствие того, что
# правит их пока код: когда потребитель появится, эталоном станет этот файл, а `default_profile()`
# останется запасным значением на случай бандла без секции.

# Геометрическое прощение (вертикаль 3): сдвиг вбок из-под угла потолка на подъёме и притяжение к
# полу на спуске. Ноль выключает приём целиком. Потолок у обоих — один тайл (16): сдвиг шире тайла
# переносит персонажа сквозь тот угол, который обходит, а притяжение глубже тайла стягивает обратно
# того, кто сошёл с края намеренно.
corner_correction | 4     # четверть тайла: прощаем задетый угол, а не промах в соседний проход
ground_snap       | 8     # полтайла: мерит высоту ступеньки под ногами, а не ширину промаха

# Проходимость наклонной грани — ОТНОШЕНИЕМ подъёма к пробегу, не углом и не косинусом: тайл-склон
# выражает ровно 45°, а косинус 45° в Q16.16 — округлённое число, и порог, записанный им, решал бы
# судьбу самой частой грани разрядом округления. Единица пропускает 45° по равенству, ноль
# оставляет полом только идеально ровное, потолок — 2 (63.43°), и он выведен из притяжения к полу:
# на более крутом спуске персонаж уходит вниз дальше `ground_snap` за тик и теряет опору.
max_slope         | 1     # 45°: ровно то, что умеет выложить карта тайлами-склонами
