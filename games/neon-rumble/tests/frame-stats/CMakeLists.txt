cmake_minimum_required(VERSION 3.21)
project(rumble_frame_stats_fixture CXX)
find_package(like-nes 0.1 REQUIRED CONFIG)
get_filename_component(GAME_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${GAME_ROOT}/frame_stats.cmake")
add_executable(rumble_frame_stats_fixture driver.cpp)
target_include_directories(rumble_frame_stats_fixture PRIVATE "${GAME_ROOT}/src")
target_link_libraries(rumble_frame_stats_fixture PRIVATE like-nes::engine)
if(MSVC)
  target_compile_options(rumble_frame_stats_fixture PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(rumble_frame_stats_fixture PRIVATE -Wall -Wextra -Werror -Wold-style-cast)
endif()
like_nes_add_game(rumble_frame_stats_fixture)
rumble_frame_stats_sources(rumble_frame_stats_fixture)
like_nes_bake(rumble_frame_stats_fixture MANIFEST "${GAME_ROOT}/bench/frame-stats.manifest" OUT frame-stats.bundle)
